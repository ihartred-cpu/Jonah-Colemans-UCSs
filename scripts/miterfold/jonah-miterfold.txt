			delete AA_UCS_SET_MITB
		end if
		if this.AA_UCS_SET_MITR == 1 then
			delete MITR
			delete AA_UCS_SET_MITR
		end if
		if this.AA_UCS_SET_MITT == 1 then
			delete MITT
			delete AA_UCS_SET_MITT
		end if
		if this.AA_UCS_SET_MITL == 1 then
			delete MITL
			delete AA_UCS_SET_MITL
		end if
	 			
 		TEMP_Edge<int> := 1
 		while TEMP_Edge <= TEMP_Number_Edges do
 			if this.AA_UCS_SET_EDG{TEMP_Edge}EXT != null then
 				delete _EDG{TEMP_Edge}EXT
 				delete AA_UCS_SET_EDG{TEMP_Edge}EXT
 				delete _EXTEDX
 				delete _EXTEDY
 			end if
 			if this.AA_MiterFold_Edge{TEMP_Edge}_Angle != null or this.AA_MiterFold_Edge{TEMP_Edge}_Size != null then
	 			delete AA_MiterFold_Edge{TEMP_Edge}_Angle
	 			delete AA_MiterFold_Edge{TEMP_Edge}_Size
	 			delete AA_MiterFold_Edge{TEMP_Edge}_Miter
	 			delete AA_MiterFold_Edge{TEMP_Edge}_ExtraExt
	 			delete AA_MiterFold_Edge{TEMP_Edge}_SeparatePartNew
	 			delete AA_MiterFold_Edge{TEMP_Edge}_Offset
	 		end if
 			
 			TEMP_Edge<int> += 1
 		end while
 		delete TEMP_Edge
 	end if
 	
 	delete TEMP_Number_Edges
 	
 	exit
