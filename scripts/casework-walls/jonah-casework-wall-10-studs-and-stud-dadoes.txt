 ;CV2025-PATCH (this repo, not Jonah Coleman's): two fixes below for a CV 2025.3 regression reported and
 ;confirmed (Verified Answer) on the Hexagon Nexus forum -- "_EDGWP now used instead of _FACEWP on
 ;non-shaped parts - intentional?" (Wes Shimwell / Toby Richards). See docs/casework-walls-jonah.md.
 ;  1) line ~30: the "is this part shaped" test read _EDGWP == 0, which CV 2025.3 can now report
 ;     non-zero on a part that was never shaped (a side effect of per-edge connection dimming).
 ;     Swapped to _SHPEDGCNT > 0, the shape-count parameter Toby Richards gives as the correct test.
 ;  2) line ~366 (was ~365): DY -= TEMP_TopAdjust could silently shave ~0.012mm off DY from residual
 ;     floating-point error even when TEMP_TopAdjust "reads" as 0, which stopped downstream top-plate
 ;     dado operations from firing (they weren't in real geometric contact anymore). Wrapped the
 ;     subtracted value in rprec() so a near-zero adjustment rounds to exactly zero before it's applied.
 ;These are sourced from a corroborated forum report, not run against a real CV2025 install by this repo --
 ;see TESTING.md before trusting them on a production job.

												end if
											end if

												CLEATSCREW.SPCNG := TEMP_CLEATDADO_DY/3
											else
												if TEMP_CLEATDADO_DY > 8 then
													CLEATSCREW.Y := DY - TEMP_CLEATDADO_Y - TEMP_CLEATDADO_DY/4
													CLEATSCREW.REPT := 3
													CLEATSCREW.SPCNG := TEMP_CLEATDADO_DY/4
												end if
											end if
											
											;need to adjust top/bottom holes so they don't interfere with top/bottom screws
											if TEMP_CLEATDADO_Y + TEMP_CLEATDADO_DY > DY - 1 then
												;at the top
												CLEATSCREW.Y := TEMP_CLEATDADO_DY - 0.5
												CLEATSCREW.REPT := 1
											else

										else
											CLEATSCREW.Z -= AA_AdjustPosition_Origin
										end if

									end if
								end if
							
							
								delete TEMP_CLEATDADO_DY
								delete TEMP_CLEATDADO_Y
							end if
							
							;AA_CaseworkWall_Signature<text> = '{AA_CaseworkWall_Signature}NOTCH{TEMP_Loop_Notches_Inner}-{TEMP_Loop1}:DY={TEMP_CLEATDADO_DY},Y={TEMP_CLEATDADO_Y},DX={TEMP_Notch_DX},PLOW={TEMP_Notch_PLOW},FACE={TEMP_Loop_Face},INSET={TEMP_Notch_INSET};'
							
							TEMP_Loop_Notches_Inner<int> += 1
						end while
					end if
					
					TEMP_Loop1<int> += 1
				end while
				if this.AA_DisableCleats_{TEMP_Loop_Face}_{TEMP_Loop1} != null then

					while this.AA_DisableCleats_{TEMP_Loop_Face}_{TEMP_Loop1} != null do

						TEMP_Loop1<int> += 1
					end while

				delete TEMP_Loop1
				
				delete TEMP_Notch_DX
				delete TEMP_Notch_AFF
				delete TEMP_Notch_DY
				delete TEMP_Notch_Inset
				delete TEMP_Notch_Desc
				delete TEMP_Notch_PLOW
				delete TEMP_Notch_Screw
				
				delete TEMP_Loop_Notches_Inner
				delete TEMP_Loop_Notches_Inner_End

				delete TEMP_LAST_NOTCH_TOP_AFF
				
				;wire chases
				if AA_DisableWireChase == 0 then
					TEMP_Loop1<int> := 1
					while TEMP_Loop1 <= Cab.{Use_FaceName}.TR.AA_WireChase_Number or this.AA_WireChase_{TEMP_Loop1}_Ignore != null do
						if TEMP_Loop1 > Cab.{Use_FaceName}.TR.AA_WireChase_Number then
							delete AA_WireChase_{TEMP_Loop_Face}_{TEMP_Loop1}_Ignore  
						else
							if this.AA_WireChase_{TEMP_Loop_Face}_{TEMP_Loop1}_Ignore   == null then
								AA_WireChase_{TEMP_Loop_Face}_{TEMP_Loop1}_Ignore<bool> = false
							end if
							AA_WireChase_{TEMP_Loop_Face}_{TEMP_Loop1}_Ignore<style> = AA_Stud_OptionState
							if TEMP_Loop1_Face == 1 then
								AA_WireChase_{TEMP_Loop_Face}_{TEMP_Loop1}_Ignore<desc> = 'Wall: 06) Disable Wire Chase: Attach Face #{TEMP_Loop1}'
							else
								AA_WireChase_{TEMP_Loop_Face}_{TEMP_Loop1}_Ignore<desc> = 'Wall: 06) Disable Wire Chase: Opp Face #{TEMP_Loop1}'
							end if
							
							if AA_WireChase_{TEMP_Loop_Face}_{TEMP_Loop1}_Ignore == 0 then
								TEMP_WireChase_Width := Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_Width
								if TEMP_WireChase_Width == -1 then
									if DX > 2 then
										if DX > 5 then
											TEMP_WireChase_Width :=  DX - 3
										else
											TEMP_WireChase_Width :=  DX - 1.5
										end if
										if TEMP_WireChase_Width > 4 then
											TEMP_WireChase_Width := 4
										end if
									else
										TEMP_WireChase_Width := 0
									end if 
								end if
								
								if TEMP_WireChase_Width > 0 then
									if Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_AFF + Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_Height/2 < DY + PABSY then  
										if Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_AFF - Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_Height/2 > PABSY then  
											if Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_Type == 1 then
												dim WireChaseRectangle as new route
												WireChaseRectangle._RCUT := 1 
												;WireChaseRectangle.TOOLID := 141 ;roughing tool
												WireChaseRectangle._FACEWP<int> := 1
												WireChaseRectangle.desc = 'Wire Chase #{TEMP_Loop1}'
												
												if Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_PosX <= 0 then
													WireChaseRectangle.X := DX/2 - TEMP_WireChase_Width/2
												else
													if TEMP_Loop_Face == 1 then
														WireChaseRectangle.X := Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_PosX
													else
														WireChaseRectangle.X := DX - Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_PosX - TEMP_WireChase_Width
													end if
												end if
												
												WireChaseRectangle.Y := Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_AFF - PABSY - Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_Height/2
												WireChaseRectangle.Z := 0
												
												WireChaseRectangle.DX := TEMP_WireChase_Width
												WireChaseRectangle.DY := Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_Height
												WireChaseRectangle.DZ := DZ
												
												WireChaseRectangle.AX := 0
												WireChaseRectangle.AY := 0
												WireChaseRectangle.AZ := 0
												
												TEMP_OUTSIDERADIUS := 1
												if TEMP_WireChase_Width < 2.25 then
													TEMP_OUTSIDERADIUS := TEMP_WireChase_Width/2 - 0.25
												end if
												WireChaseRectangle.OUTSIDERADIUS := TEMP_OUTSIDERADIUS
												delete TEMP_OUTSIDERADIUS
												
												WireChaseRectangle.LibPart = 'Jonah Rocks\Generic Shapes\Rectangle With Radius Corners'
											end if
											
											if Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_Type == 2 then
												dim WireChaseCircle as new HOLE
												WireChaseCircle._FACEWP<int> := 1
												;WireChaseCircle.TOOLID := 141 ;roughing tool
												;WireChaseCircle._POCKET := 1
												WireChaseCircle.desc = 'Wire Chase #{TEMP_Loop1}'
												
												if Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_PosX <= 0 then
													WireChaseCircle.X := DX/2
												else
													if TEMP_Loop_Face == 1 then
														WireChaseCircle.X := Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_PosX
													else
														WireChaseCircle.X := DX - Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_PosX
													end if
												end if 
												
												WireChaseCircle.Y := Cab.{Use_FaceName}.TR.AA_WireChase_{TEMP_Loop1}_AFF - PABSY 
												WireChaseCircle.Z := 0
												
												WireChaseCircle.DX := TEMP_WireChase_Width
												WireChaseCircle.DY := TEMP_WireChase_Width
												WireChaseCircle.DZ := DZ
												
												WireChaseCircle.AX := 0
												WireChaseCircle.AY := 0
												WireChaseCircle.AZ := 0
											end if
										end if
									end if
								end if
								delete TEMP_WireChase_Width
							end if
						end if

					end while
					delete TEMP_Loop1
				end if
					
				TEMP_Loop_Face<int> += 1
			end while
			delete Use_FaceName
			delete TEMP_Loop_Face
			delete TEMP_EndLoop
			
			
			;add square outs
			if this.CLEATDADO.DX != null then
				TEMP_Loop_Instance<int> := 1
				while this.CLEATDADO@{TEMP_Loop_Instance}.DX != null do
					TEMP_Top_Is_Contained<int> := 0
					TEMP_Bottom_Is_Contained<int> := 0
					
					;check to see if this dado is contained by another, so we don't drill square outs then
					if :.{name}@1.OBJID == OBJID or DY != :.{name}@1.DY or AA_DisableCleats + AA_DisableToeNotch + AA_DisableCounterNotch > 0 then
						;first one in the section, we'll do the math here - this is extremely bad for performance so we want to keep ourselves from doing it on each stud
						TEMP_Loop_Check<int> := 1
						while TEMP_Loop_Check < TEMP_Loop_Instance do ;this.CLEATDADO@{TEMP_Loop_Check}.DX != null do
							if TEMP_Loop_Check != TEMP_Loop_Instance then
								if this.CLEATDADO@{TEMP_Loop_Instance}.REF_FACE == this.CLEATDADO@{TEMP_Loop_Check}.REF_FACE then
									;same face reference
									if this.CLEATDADO@{TEMP_Loop_Instance}.DX <= this.CLEATDADO@{TEMP_Loop_Check}.DX then
										;the dado we're thinking about squaring out is less deep or equal to the one we're comparing
										if this.CLEATDADO@{TEMP_Loop_Instance}.Y >= this.CLEATDADO@{TEMP_Loop_Check}.Y then
											if this.CLEATDADO@{TEMP_Loop_Instance}.Y <= this.CLEATDADO@{TEMP_Loop_Check}.Y + this.CLEATDADO@{TEMP_Loop_Check}.DY then
												TEMP_Bottom_Is_Contained<int> := 1
											end if
										 end if
										if this.CLEATDADO@{TEMP_Loop_Instance}.Y + this.CLEATDADO@{TEMP_Loop_Instance}.DY >= this.CLEATDADO@{TEMP_Loop_Check}.Y  then
											if this.CLEATDADO@{TEMP_Loop_Instance}.Y + this.CLEATDADO@{TEMP_Loop_Instance}.DY <= this.CLEATDADO@{TEMP_Loop_Check}.Y + this.CLEATDADO@{TEMP_Loop_Check}.DY then									
												TEMP_Top_Is_Contained<int> := 1
											end if

									end if

						AA_CaseworkWall_CleatSquareOut_{TEMP_Loop_Instance}_Top_Is_Contained<int> := TEMP_Top_Is_Contained

						if this.CLEATDADO@{TEMP_Loop_Instance}.Y == 0 or TEMP_Bottom_Is_Contained == 1 then
							;at bottom
							SQUARELINEBORE.Y += TEMP_SPCNG
							SQUARELINEBORE.REPT := 1
						end if
						;we should check to see if there is a >= depth notch immediately above
						if this.CLEATDADO@{TEMP_Loop_Instance}.Y + this.CLEATDADO@{TEMP_Loop_Instance}.DY >= DY or TEMP_Top_Is_Contained == 1 then
							;at top
							SQUARELINEBORE.REPT := 1
						end if
						delete TEMP_SPCNG
					end if
					
					delete TEMP_Top_Is_Contained
					delete TEMP_Bottom_Is_Contained
					
					TEMP_Loop_Instance<int> += 1
				end while
				delete TEMP_Loop_Instance
			end if
		end if
	
		if desc == 'Partition' or desc == 'Stud' then
			desc = 'Stud'
			if Cab.{TEMP_FaceName}.TR.AA_Studs_Number == 1 or Cab.{TEMP_OppFaceName}.TR.AA_Studs_Number == 1 then
				TEMP_Instance<int> := 1
				AA_StudNo<int> := 1 
				while :.PT@{TEMP_Instance}.X != null do
					if :.PT@{TEMP_Instance}.X < X then
						AA_StudNo<int> := AA_StudNo + 1
					end if
					TEMP_Instance<int> += 1
				end while
				delete TEMP_Instance
				desc = 'Stud {TEMP_FaceName} - {AA_StudNo}'
			end if
		end if
