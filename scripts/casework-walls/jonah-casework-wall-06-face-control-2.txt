					end if

				end while

				end if

					AA_RemovablePanels_DefaultWidth_Use := (:DX - TEMP_Total_Space_Taken)/(AA_RemovablePanels_Number - TEMP_Qty_Set_Widths) 
				end if

				delete TEMP_EQL_Use_DX
				delete TEMP_Qty_Set_Positions
				delete TEMP_Total_Space_Taken
				delete TEMP_Qty_Set_Widths 
			end if

				;relative positioning
				TEMP_Effective_PosX_Value := AA_RemovablePanels_{TEMP_Loop_Rel_Pos}_PosX_New 
				TEMP_USE_DX := AA_RemovablePanels_{TEMP_Loop_Rel_Pos}_Width 
				if TEMP_USE_DX <= 0 then
					TEMP_USE_DX :=  AA_RemovablePanels_DefaultWidth_Use
				end if
				
				if TEMP_Effective_PosX_Value == -1 then 
					;left/right clearances
					if TEMP_Loop_Rel_Pos == 1 and AA_RemovablePanels_LeftClearance <> -1 then
						TEMP_Effective_PosX_Value := AA_RemovablePanels_LeftClearance
					end if 
					if TEMP_Loop_Rel_Pos == AA_RemovablePanels_Number and AA_RemovablePanels_RightClearance <> -1 then
						TEMP_Effective_PosX_Value := :DX - (TEMP_Pos_X + TEMP_USE_DX) - AA_RemovablePanels_RightClearance
					end if
					
					if TEMP_Effective_PosX_Value == -1 then
						;use default
						TEMP_Effective_PosX_Value := AA_RemovablePanels_DefaultPOSX_Use
					end if
				end if
				
				AA_RemovablePanels_{TEMP_Loop_Rel_Pos}_PosX_Effective<meas> := TEMP_Pos_X + TEMP_Effective_PosX_Value
				
				TEMP_Pos_X<meaS> += TEMP_Effective_PosX_Value + TEMP_Use_DX
				delete TEMP_Effective_PosX_Value
				
				TEMP_Loop_Rel_Pos<int> += 1
			end while
			delete TEMP_Loop_Rel_Pos
		
			
			delete TEMP_Pos_X
			delete TEMP_USE_DX		
		end if
